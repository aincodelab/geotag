Simpan ke Google Drive

Simpan foto hasil ke folder Google Drive dan catat metadatanya — termasuk Deskripsi Kegiatan — ke Google Spreadsheet melalui Apps Script Web App.

Belum ada aktivitas. Tekan tombol di atas untuk mulai menyimpan.

URL hasil deploy Apps Script (Akses: Siapa saja).

Ambil dari URL spreadsheet: docs.google.com/spreadsheets/d/ID/edit

Ambil dari URL folder: drive.google.com/drive/folders/ID

Sheet akan dibuat otomatis beserta header jika belum ada.

Code.gs 
/**
 * Pro Geotag — Apps Script Web App
 * Simpan foto ke Google Drive + catat metadata (termasuk Deskripsi Kegiatan)
 * ke Google Spreadsheet.
 *
 * CARA PAKAI:
 * 1. Buka script.google.com → New Project
 * 2. Hapus isi default, paste kode ini
 * 3. Ganti SPREADSHEET_ID dan FOLDER_ID di bawah
 * 4. Deploy → New Deployment → Web App
 *    - Execute as: Me
 *    - Who has access: Anyone
 * 5. Salin URL /exec ke Pro Geotag
 */

const SPREADSHEET_ID = 'ISI_DENGAN_ID_SPREADSHEET_ANDA';
const FOLDER_ID      = 'ISI_DENGAN_ID_FOLDER_DRIVE_ANDA';
const SHEET_NAME     = 'Geotag';

const HEADERS = [
  'Timestamp','Nama File','URL File','File ID',
  'Latitude','Longitude','Koordinat','Alamat',
  'Deskripsi Kegiatan',
  'Waktu Foto','Judul','Panjang','Lebar'
];

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);

    const fileName  = data.fileName || ('geotag_' + Date.now() + '.jpg');
    const mimeType  = data.mimeType || 'image/jpeg';
    const base64    = data.base64;

    if (!base64) throw new Error('Data gambar kosong.');

    // 1. Simpan ke Google Drive
    const folder = DriveApp.getFolderById(FOLDER_ID);
    const blob = Utilities.newBlob(
      Utilities.base64Decode(base64),
      mimeType,
      fileName
    );
    const file = folder.createFile(blob);

    // 2. Simpan metadata ke Spreadsheet
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    let sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) sheet = ss.insertSheet(SHEET_NAME);

    if (sheet.getLastRow() === 0) {
      sheet.appendRow(HEADERS);
      sheet.getRange(1, 1, 1, HEADERS.length)
           .setFontWeight('bold')
           .setBackground('#0f172a')
           .setFontColor('#ffffff');
      sheet.setFrozenRows(1);
    }

    sheet.appendRow([
      new Date(),
      fileName,
      file.getUrl(),
      file.getId(),
      data.lat || '',
      data.lng || '',
      data.coords || '',
      data.address || '',
      data.description || '',   // <-- Deskripsi Kegiatan
      data.photoTime || '',
      data.title || '',
      data.width || '',
      data.height || ''
    ]);

    return ContentService
      .createTextOutput(JSON.stringify({
        success: true,
        fileUrl: file.getUrl(),
        fileId: file.getId(),
        fileName: fileName
      }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({
        success: false,
        error: String(err && err.message ? err.message : err)
      }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  return ContentService
    .createTextOutput(JSON.stringify({
      status: 'ok',
      message: 'Pro Geotag Apps Script endpoint aktif.'
    }))
    .setMimeType(ContentService.MimeType.JSON);
}

function testSetup() {
  const folder = DriveApp.getFolderById(FOLDER_ID);
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  Logger.log('Folder OK: ' + folder.getName());
  Logger.log('Spreadsheet OK: ' + ss.getName());
}

Panduan Singkat

  1. Buat Google Spreadsheet → salin ID dari URL.
  2. Buat Folder di Google Drive → salin ID dari URL.
  3. Buka script.google.com → New Project → paste kode di atas.
  4. Isi SPREADSHEET_ID & FOLDER_ID pada kode.
  5. Deploy → Web App → Akses: Anyone → salin URL /exec.
  6. Paste URL & ID di konfigurasi Pro Geotag → Test Koneksi.
  7. Isi Deskripsi Kegiatan di menu sidebar sebelum menyimpan foto.
